### 前端使用示例

```tsx
// 使用React Hook
import { useWebSocket } from '@/lib/hooks/use-websocket';

function TaskMonitor({ taskId }) {
  const { isConnected, lastMessage, subscribeToRoom } = useWebSocket({
    token: userToken,
    userId: currentUser.id,
    autoConnect: true,
    reconnectAttempts: 5,
    onMessage: (msg) => {
      console.log('收到消息:', msg.type, msg.payload);
    },
  });

  useEffect(() => {
    if (isConnected && taskId) {
      subscribeToRoom(`task:${taskId}`);
    }
  }, [isConnected, taskId]);

  return (
    <div>
      连接状态: {isConnected ? '🟢 已连接' : '🔴 断开'}
      最新消息: {lastMessage?.type}
    </div>
  );
}
```

### 安全机制矩阵

| 安全措施 | 实现方式 | 保护级别 |
|---------|---------|---------|
| **CSRF防护** | State参数 + 10分钟过期 | 🔴 Critical |

| **暴力破解防护** | 验证码限流 + 失败计数 | 🟠 High |
| **会话劫持防护** | IP/User-Agent绑定检查 | 🟠 High |
| **MFA绕过防护** | 密码二次确认禁用MFA | 🟡 Medium |
| **恢复码保护** | 一次性使用 + 数量限制 | 🟡 Medium |

```typescript
// 认证流程

     → 返回授权URL (含state参数)

     → 交换Authorization Code获取Tokens
     → 返回: tokens + userInfo (+ mfaRequired?)

// MFA管理

     → 返回: secret + QR Code URL + backupCodes

     → 验证TOTP代码

POST /api/auth/oauth { action: 'mfa/send-sms', userId, phoneNumber }
     → 发送SMS验证码

POST /api/auth/oauth { action: 'mfa/disable', userId, currentPassword }
```

### 租户解析策略

```typescript
// Tenant Middleware 解析优先级
interface TenantResolutionResult {
  tenantId: string;
  resolvedBy: 'header' | 'subdomain' | 'custom_domain' | 'default';
}

// 优先级1: X-Tenant-ID Header (API调用)
GET /api/tasks
Headers: { X-Tenant-ID: "tenant_abc123" }

// 优先级2: Custom Domain (自定义域名)
https://client-a.yourplatform.com/api/tasks
→ 解析为 tenant_a

// 优先级3: Subdomain (子域名)
https://company-b.app.yourplatform.com/api/tasks
→ 解析为 company-b

// 优先级4: Default Tenant (单租户模式)
→ 使用 DEFAULT_TENANT_ID 环境变量
```

### 数据隔离策略对比

| 策略 | 适用场景 | 优点 | 缺点 | 实现复杂度 |
|------|---------|------|------|-----------|
| **Row-Level** (默认) | 中小规模SaaS | 简单、共享资源 | 需注意SQL注入 | ⭐ 低 |
| **Schema隔离** | 大型企业客户 | 逻辑隔离清晰 | Schema管理复杂 | ⭐⭐⭐ 中高 |
| **Database隔离** | 金融/医疗等合规行业 | 物理隔离最强 | 成本高、运维复杂 | ⭐⭐⭐⭐⭐ 高 |

### 内置分析维度

#### 1️⃣ 通话分析 (Call Analytics)

| 指标 | 类型 | 聚合方式 | 说明 |
|------|------|---------|------|
| call_count | Counter | Count | 通话总次数 |
| avg_duration | Histogram | Average | 平均通话时长(秒) |
| success_rate | Gauge | Average | 成功率(%) |
| total_duration | Counter | Sum | 总时长(分钟) |
| cost_per_call | Gauge | Average | 单次成本(元) |

#### 2️⃣ 任务分析 (Task Analytics)

| 指标 | 类型 | 聚合方式 | 说明 |
|------|------|---------|------|
| total_tasks | Counter | Count | 任务总数 |
| completed_tasks | Counter | Count | 已完成任务数 |
| completion_rate | Gauge | Average | 完成率(%) |
| avg_processing_time | Histogram | Average | 平均处理时间(分) |
| error_rate | Gauge | Average | 错误率(%) |

#### 3️⃣ 用户活跃度 (User Activity)

| 指标 | 类型 | 聚合方式 | 说明 |
|------|------|---------|------|
| active_users | Gauge | DistinctCount | 活跃用户数(DAU/MAU) |
| api_calls | Counter | Sum | API调用次数 |
| avg_session_duration | Histogram | Average | 平均会话时长(分) |
| new_users | Counter | Count | 新增用户数 |
| retention_rate | Gauge | Average | 留存率(%) |

```typescript
// 获取通话分析数据
const result = await analyticsEngine.getCallAnalytics({
  timeRange: {
    start: new Date(Date.now() - 7 * 24 * 60 * 60 * 1000), // 最近7天
    end: new Date(),
    granularity: TimeGranularity.DAILY,
  },
});

// 结果包含:
// - timeSeriesData: 每日的通话次数/时长/成功率
// - trend: 上升趋势 +8.5%, 未来7天预测
// - metadata: 执行时间、记录总数

// 创建自定义仪表盘
const dashboard = await dashboardManager.createDashboard({
  name: '销售团队监控面板',
  ownerId: 'user_123',
});

await dashboardManager.addWidget(dashboard.id, {
  type: 'line_chart',
  title: '月度转化率趋势',
  query: {
    metrics: [{
      field: 'conversion_rate',
      label: '转化率',
      type: MetricType.GAUGE,
      aggregation: AggregationType.AVERAGE,
    }],
    timeRange: {
      start: new Date('2026-01-01'),
      end: new Date(),
      granularity: TimeGranularity.MONTHLY,
    },
  },
  config: {
    showLegend: true,
    showGrid: true,
    format: 'percentage',
  },
  position: { x: 0, y: 0, width: 12, height: 6 },
});

// 导出CSV
const exportResult = await analyticsEngine.exportData(query, 'csv');
// → filename: analytics_20260502.csv
// → data: CSV字符串
```
