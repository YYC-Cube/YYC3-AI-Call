// 医疗领域示例:
const medicalKG = new KnowledgeGraphEngine('medical');

medicalKG.extractEntities("患者张三患有2型糖尿病，正在服用二甲双胍");
// 输出: [
//   { type: "Patient", name: "张三", id: "PAT001" },

//   { type: "Drug", name: "二甲双胍", atc: "A10BA02" }

medicalKG.inferRelations(
  { id: "DIS001" }, // 2型糖尿病
  { id: "DRG001" } // 二甲双胍
);
// 输出: [{ relation: "treats", confidence: 0.98 }]
