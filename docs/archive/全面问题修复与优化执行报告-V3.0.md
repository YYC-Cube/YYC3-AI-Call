private cleanupTaskQueue(): void {
  const cutoffTime = Date.now() - this.TASK_RETENTION_MS;
  
  // 移除超过1小时的已完成/失败任务
  this.taskQueue = this.taskQueue.filter(task => {

}

```typescript
export class RequestDeduplicator {
  // 基于内容哈希的去重键生成
  private generateRequestHash(request, body): string {
    const data = {
      method: request.method,
      url: request.url.pathname,
      userId: request.headers.get('x-user-id'),
      bodyHash: hashObject(body),  // MD5哈希
    };
    return crypto.createHash('sha256')
      .update(JSON.stringify(data))
      .digest('hex');
  }

  // 检查是否重复请求
  async check(request, body): Promise<{
    isDuplicate: boolean;
    requestId: string;
    retryAfterMs: number;
  }> {
    const dedupKey = this.generateRequestHash(request, body);
    
    // Redis/内存存储 + TTL过期
    const count = await store.increment(dedupKey, ttlSeconds);
    
    return {
      isDuplicate: count > maxRequests,  // 默认允许1次
      requestId: crypto.randomUUID(),
      retryAfterMs: windowMs
    };
  }
}

// 中间件集成
export async function deduplicationMiddleware(request) {
  // GET/HEAD请求跳过去重
  if (request.method === 'GET') return { allowed: true };
  
  // 提取请求体计算哈希
  const body = await request.clone().json();
  const result = await deduplicator.check(request, body);
  
  if (result.isDuplicate) {
    return {
      allowed: false,
      errorResponse: NextResponse.json(
        { error: 'Duplicate request detected', code: 'E0007' },
        { status: 429, headers: {'Retry-After': '5'} }
      )
    };
  }
  
  return { allowed: true };
}
```

**去重特性:**
```
✅ 基于内容的智能去重 (相同参数视为重复)
✅ 可配置时间窗口 (默认5秒)
✅ 内存存储 + 自动过期
✅ 排除GET等安全请求
✅ 支持自定义排除路径

export class ErrorSanitizer {
  // 敏感信息模式库
  private SENSITIVE_PATTERNS = [
    /password/i, /secret/i, /api[_-]?key/i,
    /token/i, /credential/i, /authorization/i
  ];
  
  private CONNECTION_STRING_PATTERNS = [
    /mongodb(?:\+srv)?:\/\/[^@]+@/,
    /postgresql?:\/\/[^@]+@/
  ];

  sanitize(error, options): SanitizedError {
    let message = error.message;
    
    if (options.maskSensitiveData) {
      // 1. 屏蔽敏感字段值
      message = this.maskSensitiveInfo(message);
      // 例: password=abc123 → password=***REDACTED***
      
      // 2. 屏蔽连接字符串
      message = this.maskConnectionStrings(message);
      // 例: mongodb://user:pass@host → mongodb://***:***@host
      
      // 3. 截断过长消息 (500字符上限)
      message = message.substring(0, 500);
    }

    return {
      message,
      code: this.categorizeError(error),  // E0001-E0103
      requestId: crypto.randomUUID(),
      timestamp: new Date().toISOString(),
      ...(options.includeStack && { stack: sanitizedStack })
    };
  }

  // API响应专用格式
  sanitizeForApiResponse(error, statusCode) {
    const sanitized = this.sanitize(error, {
      includeStack: false,       // 生产环境不返回堆栈
      maskSensitiveData: true,   // 屏蔽敏感信息
    });
    
    return {
      success: false,
      error: statusCode >= 500 && !sanitized.isOperational
        ? 'An unexpected error occurred'  // 通用错误消息
        : sanitized.message,              // 业务错误显示详情
      code: sanitized.code,
      requestId: sanitized.requestId
    };
  }
}
```
